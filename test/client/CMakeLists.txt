cmake_minimum_required(VERSION 4.0)
project(mka-module-client LANGUAGES CXX)
# BMI and client must use the same standard dialect.
set(CMAKE_CXX_EXTENSIONS OFF)
set(MKA_BUILD_TESTS OFF CACHE BOOL "Client must not depend on GoogleTest" FORCE)
if(MKA_SOURCE_DIR)
    add_subdirectory("${MKA_SOURCE_DIR}" producer)
    set(mka_client_core mka-core-audio)
else()
    find_package(mkaAudioCore CONFIG REQUIRED COMPONENTS ${MKA_CLIENT_COMPONENTS})
    set(mka_client_core mka::core)
endif()
enable_testing()
add_executable(client-core core.cpp)
target_link_libraries(client-core PRIVATE ${mka_client_core})
add_test(NAME client.core COMMAND client-core)
if(TARGET mka-dsp-example)
    add_test(NAME client.dsp_example COMMAND mka-dsp-example)
endif()
foreach(backend alsa jack pipewire pulseaudio wasapi coreaudio)
    if(TARGET mka-audio-${backend})
        set(native mka-audio-${backend})
    elseif(TARGET mka::${backend})
        set(native mka::${backend})
    else()
        continue()
    endif()
    if(native)
        add_executable(client-${backend} backend.cpp)
        target_compile_definitions(client-${backend} PRIVATE MKA_CLIENT_${backend}=1)
        target_link_libraries(client-${backend} PRIVATE ${native})
        add_test(NAME client.${backend} COMMAND client-${backend})
    endif()
endforeach()
