cmake_minimum_required(VERSION 4.0)
project(mka-module-client LANGUAGES CXX)
set(MKA_BUILD_TESTS OFF CACHE BOOL "Client must not depend on GoogleTest" FORCE)
add_subdirectory("${MKA_SOURCE_DIR}" producer)
enable_testing()
add_executable(client-core core.cpp)
target_link_libraries(client-core PRIVATE mka-core-audio)
add_test(NAME client.core COMMAND client-core)
foreach(backend alsa jack pipewire pulseaudio wasapi coreaudio)
    if(TARGET mka-audio-${backend})
        add_executable(client-${backend} backend.cpp)
        target_compile_definitions(client-${backend} PRIVATE MKA_CLIENT_${backend}=1)
        target_link_libraries(client-${backend} PRIVATE mka-audio-${backend})
        add_test(NAME client.${backend} COMMAND client-${backend})
    endif()
endforeach()
